```python
# apps/api/src/models/recipe.py
"""Recipe models — immutable Airlock defaults and workspace editable copies."""

from datetime import datetime

from sqlalchemy import DateTime, Integer, String, Text, func
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import Mapped, mapped_column

from src.db import Base


class AirlockRecipe(Base):
    __tablename__ = "airlock_recipes"

    id: Mapped[str] = mapped_column(Text, primary_key=True)
    vault_type: Mapped[str] = mapped_column(String(100), nullable=False)
    chamber: Mapped[str] = mapped_column(String(50), nullable=False)   # discover|build|review|ship
    role: Mapped[str] = mapped_column(String(50), nullable=False)      # builder|gatekeeper|owner|conductor|viewer
    version: Mapped[int] = mapped_column(Integer, default=1, nullable=False)
    nodes: Mapped[list] = mapped_column(JSONB, server_default="[]", nullable=False)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now())


class WorkspaceRecipe(Base):
    __tablename__ = "workspace_recipes"

    id: Mapped[str] = mapped_column(Text, primary_key=True)
    workspace_id: Mapped[str] = mapped_column(Text, nullable=False)    # FK → workspaces
    vault_type: Mapped[str] = mapped_column(String(100), nullable=False)
    chamber: Mapped[str] = mapped_column(String(50), nullable=False)
    role: Mapped[str] = mapped_column(String(50), nullable=False)
    sourced_from: Mapped[str | None] = mapped_column(Text, nullable=True)  # FK → airlock_recipes.id
    nodes: Mapped[list] = mapped_column(JSONB, server_default="[]", nullable=False)
    metadata_: Mapped[dict] = mapped_column("metadata", JSONB, server_default="{}", nullable=False)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now())
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), server_default=func.now(), onupdate=func.now()
    )
    updated_by: Mapped[str | None] = mapped_column(Text, nullable=True)  # FK → users
    deleted_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
```

```python
# apps/api/src/services/recipe.py
"""Recipe service — workspace recipe management and reset-to-default."""

from python_ulid import ULID
from sqlalchemy.orm import Session

from src.models.recipe import AirlockRecipe, WorkspaceRecipe


def seed_workspace_recipes(db: Session, workspace_id: str, vault_types: list[str]) -> list[WorkspaceRecipe]:
    """Copy Airlock default recipes into workspace for selected vault types.
    Called on workspace creation or when new vault types are enabled.
    Skips vault types already seeded (idempotent).
    """
    seeded = []
    airlock_recipes = (
        db.query(AirlockRecipe)
        .filter(AirlockRecipe.vault_type.in_(vault_types))
        .all()
    )
    for ar in airlock_recipes:
        existing = db.query(WorkspaceRecipe).filter_by(
            workspace_id=workspace_id,
            vault_type=ar.vault_type,
            chamber=ar.chamber,
            role=ar.role,
            deleted_at=None,
        ).first()
        if existing:
            continue
        wr = WorkspaceRecipe(
            id=str(ULID()),
            workspace_id=workspace_id,
            vault_type=ar.vault_type,
            chamber=ar.chamber,
            role=ar.role,
            sourced_from=ar.id,
            nodes=ar.nodes,
        )
        db.add(wr)
        seeded.append(wr)
    db.commit()
    return seeded


def get_recipe(db: Session, workspace_id: str, vault_type: str, chamber: str, role: str) -> WorkspaceRecipe | None:
    return db.query(WorkspaceRecipe).filter_by(
        workspace_id=workspace_id,
        vault_type=vault_type,
        chamber=chamber,
        role=role,
        deleted_at=None,
    ).first()


def update_recipe_nodes(db: Session, recipe_id: str, nodes: list, updated_by: str) -> WorkspaceRecipe:
    recipe = db.query(WorkspaceRecipe).filter_by(id=recipe_id).first()
    if not recipe:
        raise ValueError(f"Recipe {recipe_id} not found")
    recipe.nodes = nodes
    recipe.updated_by = updated_by
    db.commit()
    db.refresh(recipe)
    return recipe


def reset_recipe_to_default(db: Session, recipe_id: str, updated_by: str) -> WorkspaceRecipe:
    """Replace workspace recipe nodes with the original Airlock defaults."""
    recipe = db.query(WorkspaceRecipe).filter_by(id=recipe_id).first()
    if not recipe or not recipe.sourced_from:
        raise ValueError(f"Recipe {recipe_id} not found or has no Airlock source")
    airlock = db.query(AirlockRecipe).filter_by(id=recipe.sourced_from).first()
    if not airlock:
        raise ValueError(f"Airlock default recipe {recipe.sourced_from} not found")
    recipe.nodes = airlock.nodes
    recipe.updated_by = updated_by
    db.commit()
    db.refresh(recipe)
    return recipe


def list_recipes(db: Session, workspace_id: str, vault_type: str | None = None) -> list[WorkspaceRecipe]:
    q = db.query(WorkspaceRecipe).filter_by(workspace_id=workspace_id, deleted_at=None)
    if vault_type:
        q = q.filter_by(vault_type=vault_type)
    return q.all()
```

```python
# apps/api/src/services/invite.py
"""Invite token service — generate and redeem single-use role invite links."""

from datetime import datetime, timezone

from python_ulid import ULID
from sqlalchemy.orm import Session

from src.models.invite import InviteToken
from src.models.user_module_role import UserModuleRole


def create_invite(
    db: Session,
    workspace_id: str,
    module: str,
    role: str,
    vault_types: list[str],
    created_by: str,
) -> InviteToken:
    token = InviteToken(
        id=str(ULID()),          # ULID is the URL token — no separate secret needed
        workspace_id=workspace_id,
        module=module,
        role=role,
        vault_types=vault_types,
        created_by=created_by,
        is_active=True,
    )
    db.add(token)
    db.commit()
    db.refresh(token)
    return token


def get_invite(db: Session, token_id: str) -> InviteToken | None:
    return db.query(InviteToken).filter_by(id=token_id, is_active=True, redeemed_by=None).first()


def redeem_invite(db: Session, token_id: str, user_id: str) -> InviteToken:
    """Redeem a single-use invite token. Assigns module role to user."""
    token = get_invite(db, token_id)
    if not token:
        raise ValueError("Invite token is invalid or already used")

    # Assign module role
    existing_role = db.query(UserModuleRole).filter_by(
        user_id=user_id,
        workspace_id=token.workspace_id,
        module_type=token.module,
    ).first()
    if not existing_role:
        db.add(UserModuleRole(
            id=str(ULID()),
            user_id=user_id,
            workspace_id=token.workspace_id,
            module_type=token.module,
            role=token.role,
        ))

    # Burn the token
    token.redeemed_by = user_id
    token.redeemed_at = datetime.now(timezone.utc)
    token.is_active = False

    db.commit()
    db.refresh(token)
    return token
```

```python
from pydantic import BaseModel
from sqlalchemy.orm import Session

from src.db import get_db
from src.middleware.auth import get_current_user
from src.services.recipe import get_recipe, list_recipes, reset_recipe_to_default, update_recipe_nodes

router = APIRouter(prefix="/api/v1/recipes", tags=["recipes"])


class UpdateNodesRequest(BaseModel):
    nodes: list


@router.get("")
def list_workspace_recipes(
    vault_type: str | None = None,
    current_user=Depends(get_current_user),
    db: Session = Depends(get_db),

    ).first()

    return _to_dict(recipe)

def update_nodes(
    recipe_id: str,
    body: UpdateNodesRequest,
    current_user=Depends(get_current_user),
    db: Session = Depends(get_db),
):
    try:
        recipe = update_recipe_nodes(db, recipe_id, body.nodes, current_user.id)
        return _to_dict(recipe)
    except ValueError as e:
        raise HTTPException(404, str(e))


@router.post("/{recipe_id}/reset")

    try:
        recipe = reset_recipe_to_default(db, recipe_id, current_user.id)
        return _to_dict(recipe)
    except ValueError as e:
        raise HTTPException(404, str(e))


def _to_dict(r):
```

```python
# apps/api/src/routes/invites.py
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from sqlalchemy.orm import Session

from src.db import get_db
from src.middleware.auth import get_current_user
from src.services.invite import create_invite, get_invite, redeem_invite

router = APIRouter(prefix="/api/v1/invites", tags=["invites"])


class CreateInviteRequest(BaseModel):
    module: str
    role: str
    vault_types: list[str]


class RedeemInviteRequest(BaseModel):
    token_id: str


@router.post("")
def create_invite_route(
    body: CreateInviteRequest,
    current_user=Depends(get_current_user),
    db: Session = Depends(get_db),
):
    token = create_invite(
        db,
        workspace_id=current_user.workspace_id,
        module=body.module,
        role=body.role,
        vault_types=body.vault_types,
        created_by=current_user.id,
    )
    return {"token_id": token.id, "url": f"/invite/{token.id}"}


@router.get("/{token_id}/preview")
def preview_invite(token_id: str, db: Session = Depends(get_db)):
    """Public endpoint — no auth. Returns invite metadata for the landing page."""
    token = get_invite(db, token_id)
    if not token:
        raise HTTPException(404, "Invite not found or already used")
    return {
        "workspace_id": token.workspace_id,
        "module": token.module,
        "role": token.role,
        "vault_types": token.vault_types,
    }


@router.post("/{token_id}/redeem")
def redeem_invite_route(
    token_id: str,
    current_user=Depends(get_current_user),
    db: Session = Depends(get_db),
):
    try:
        token = redeem_invite(db, token_id, current_user.id)
        return {"redeemed": True, "module": token.module, "role": token.role}
    except ValueError as e:
        raise HTTPException(400, str(e))
```

```tsx
    fetch(`/api/v1/invites/${tokenId}/preview`)
      .then((r) => (r.ok ? r.json() : Promise.reject("Invalid or used invite")))
      .then(setPreview)
      .catch((e) => setError(typeof e === "string" ? e : "This invite is no longer valid."));
  }, [tokenId]);

  async function handleAccept() {
    setRedeeming(true);
    const res = await fetch(`/api/v1/invites/${tokenId}/redeem`, { method: "POST" });
    if (res.ok) {
      router.push("/");
    } else {
      setError("Could not redeem invite. It may have already been used.");
      setRedeeming(false);
    }
  }

  if (error) {
    return (
      <div className="flex min-h-screen items-center justify-center bg-surface-base">
        <div className="rounded-lg border border-border-subtle bg-surface-raised p-8 text-center">
          <p className="text-text-danger">{error}</p>
        </div>
      </div>
    );
  }

  if (!preview) {
    return (
      <div className="flex min-h-screen items-center justify-center bg-surface-base">
        <p className="text-text-muted">Loading invite…</p>
      </div>
    );
  }

  return (
    <div className="flex min-h-screen items-center justify-center bg-surface-base">
      <div className="w-full max-w-md rounded-lg border border-border-subtle bg-surface-raised p-8">
        <h1 className="mb-2 text-xl font-semibold text-text-primary">You've been invited</h1>
        <p className="mb-6 text-text-muted">
          Join as <span className="font-medium text-text-primary">{ROLE_LABELS[preview.role]}</span>{" "}
          in the <span className="font-medium text-text-primary capitalize">{preview.module}</span> module.
        </p>
        <div className="mb-6 rounded border border-border-subtle bg-surface-base p-4">
          <p className="mb-1 text-xs font-medium uppercase tracking-wide text-text-muted">Vault types</p>
          <ul className="space-y-1">
            {preview.vault_types.map((vt) => (
              <li key={vt} className="text-sm text-text-secondary capitalize">
                {vt.replace(/_/g, " ")}
              </li>
            ))}
          </ul>
        </div>
        <button
          onClick={handleAccept}
          disabled={redeeming}
          className="w-full rounded bg-accent-primary px-4 py-2 text-sm font-medium text-white disabled:opacity-50"
        >
          {redeeming ? "Joining…" : "Accept Invite"}
        </button>
      </div>
    </div>
  );
}
```

```tsx
// apps/web/src/components/organisms/RecipeEditor.tsx
"use client";

import { useEffect, useState } from "react";

interface RecipeNode {
  id: string;
  type: string;
  label: string;
  skills: string[];
  gate_conditions: unknown[];
  position: number;
}

interface Recipe {
  id: string;
  vault_type: string;
  chamber: string;
  role: string;
  nodes: RecipeNode[];
  updated_at: string | null;
  updated_by: string | null;
}

interface Props {
  module: string;
  vaultType: string;
  chamber: string;
  role: string;
}

export function RecipeEditor({ module, vaultType, chamber, role }: Props) {
  const [recipe, setRecipe] = useState<Recipe | null>(null);
  const [selected, setSelected] = useState<RecipeNode | null>(null);

  useEffect(() => {
    fetch(`/api/v1/recipes?vault_type=${vaultType}`)
      .then((r) => r.json())
      .then((recipes: Recipe[]) => {
        const match = recipes.find((r) => r.chamber === chamber && r.role === role);
        setRecipe(match ?? null);
      });
  }, [vaultType, chamber, role]);

  async function handleReset() {
    if (!recipe) return;
    if (!confirm("Reset to Airlock default? This cannot be undone.")) return;
    const res = await fetch(`/api/v1/recipes/${recipe.id}/reset`, { method: "POST" });
    if (res.ok) setRecipe(await res.json());
  }

  if (!recipe) return <p className="p-8 text-text-muted">Loading recipe…</p>;

  return (
    <div className="p-8">
      <div className="mb-6 flex items-center justify-between">
        <div>
          <p className="text-xs uppercase tracking-wide text-text-muted capitalize">
            {module} › {vaultType.replace(/_/g, " ")} › {chamber} › {role}
          </p>
          {recipe.updated_at && (
            <p className="text-xs text-text-muted">
              Last edited {new Date(recipe.updated_at).toLocaleDateString()}
            </p>
          )}
        </div>
        <button onClick={handleReset} className="text-xs text-text-danger hover:underline">
          Reset to Airlock Default
        </button>
      </div>

      {/* Node flow */}
      <div className="flex items-start gap-4 overflow-x-auto pb-4">
        {recipe.nodes
          .sort((a, b) => a.position - b.position)
          .map((node) => (
            <button
              key={node.id}
              onClick={() => setSelected(node)}
              className={`min-w-[140px] rounded-lg border p-4 text-left transition-colors ${
                selected?.id === node.id
                  ? "border-accent-primary bg-surface-active"
                  : "border-border-subtle bg-surface-raised hover:border-border-default"
              }`}
            >
              <p className="mb-1 text-xs font-medium uppercase tracking-wide text-text-muted">
                {node.position}
              </p>
              <p className="text-sm font-medium text-text-primary">{node.label}</p>
              <p className="mt-2 text-xs text-text-muted capitalize">{node.type}</p>
              <p className="text-xs text-text-muted">Skills {node.skills.length}</p>
              <p className="text-xs text-text-muted">Gates {node.gate_conditions.length}</p>
            </button>
          ))}
        <button className="flex min-w-[44px] items-center justify-center rounded-lg border border-dashed border-border-subtle bg-surface-base p-4 text-text-muted hover:border-border-default">
          +
        </button>
      </div>

      {/* Node detail panel */}
      {selected && (
        <div className="mt-6 rounded-lg border border-border-subtle bg-surface-raised p-6">
          <h3 className="mb-4 font-medium text-text-primary">{selected.label}</h3>
          <div className="mb-4">
            <p className="mb-2 text-xs font-medium uppercase tracking-wide text-text-muted">Skills</p>
            <ul className="space-y-1">
              {selected.skills.map((s) => (

          </div>
          <div>
            <p className="mb-2 text-xs font-medium uppercase tracking-wide text-text-muted">Gate Conditions</p>
            {selected.gate_conditions.length === 0 ? (
              <p className="text-sm text-text-muted">No conditions — node always passable</p>
            ) : (
              <ul className="space-y-1">
                {selected.gate_conditions.map((c, i) => (
                  <li key={i} className="font-mono text-xs text-text-secondary">
                    {JSON.stringify(c)}
                  </li>
                ))}
              </ul>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
```

```tsx
// apps/web/src/components/organisms/TaskRunner.tsx
"use client";

import { useEffect, useState } from "react";

interface RecipeNode {
  id: string;
  type: string;
  label: string;
  skills: string[];
  gate_conditions: unknown[];
  position: number;
}

interface Props {
  workspaceId: string;
  vaultType: string;
  chamber: string;
  role: string;
}

export function TaskRunner({ workspaceId, vaultType, chamber, role }: Props) {
  const [nodes, setNodes] = useState<RecipeNode[]>([]);
  const [currentIndex, setCurrentIndex] = useState(0);

  useEffect(() => {
    fetch(`/api/v1/recipes?vault_type=${vaultType}`)
      .then((r) => r.json())
      .then((recipes: { chamber: string; role: string; nodes: RecipeNode[] }[]) => {
        const match = recipes.find((r) => r.chamber === chamber && r.role === role);
        if (match) setNodes(match.nodes.sort((a, b) => a.position - b.position));
      });
  }, [vaultType, chamber, role]);

  const current = nodes[currentIndex];
  const total = nodes.length;
  const progress = total > 0 ? Math.round(((currentIndex) / total) * 100) : 0;

  if (!current) return <p className="p-4 text-sm text-text-muted">No recipe for this vault type.</p>;

  return (
    <div className="flex h-full flex-col p-4">
      {/* Header */}
      <div className="mb-4">
        <p className="text-xs text-text-muted capitalize">{vaultType.replace(/_/g, " ")}</p>
        <p className="text-xs text-text-muted capitalize">
          {chamber} · Step {currentIndex + 1} of {total}
        </p>
        <div className="mt-2 h-1 w-full rounded-full bg-surface-active">
          <div
            className="h-1 rounded-full bg-accent-primary transition-all"
            style={{ width: `${progress}%` }}
          />
        </div>
      </div>

      {/* Current node */}
      <div className="flex-1">
        <p className="mb-1 text-xs font-medium uppercase tracking-wide text-text-muted">{current.type}</p>
        <h3 className="mb-3 text-base font-semibold text-text-primary">{current.label}</h3>

        {/* Actions (placeholder — real actions vary by node type) */}
        <div className="mb-4 flex flex-col gap-2">
          <button className="rounded bg-accent-primary px-4 py-2 text-sm font-medium text-white">
            Continue →
          </button>
        </div>

        {/* Skills */}
        {current.skills.length > 0 && (
          <div>
            <p className="mb-2 text-xs font-medium uppercase tracking-wide text-text-muted">
              Available Skills
            </p>
            <ul className="space-y-1">
              {current.skills.map((s) => (
                <li key={s} className="text-sm text-text-secondary">
                  ⚡ {s}
                </li>
              ))}
            </ul>
          </div>
        )}
      </div>

      {/* Navigation (dev only — real nav driven by gate conditions) */}
      <div className="flex justify-between border-t border-border-subtle pt-4">
        <button
          onClick={() => setCurrentIndex((i) => Math.max(0, i - 1))}
          disabled={currentIndex === 0}
          className="text-xs text-text-muted disabled:opacity-30"
        >
          ← Back
        </button>
        <button
          onClick={() => setCurrentIndex((i) => Math.min(total - 1, i + 1))}
          disabled={currentIndex === total - 1}
          className="text-xs text-text-muted disabled:opacity-30"
        >
          Next →
        </button>
      </div>
    </div>
  );
}
```
